                              เริ่มต้นอ่าน

                              แผนที่โค้ดรายเส้น API ทุกเส้นบอกว่าใช้โค้ดอะไร อยู่ไฟล์ไหน บรรทัดไหน ตั้งแต่ Frontend, Backend ไปจนถึง Database

                              ทุกคนอ่านไฟล์นี้ก่อน มีเรื่องที่ทั้งทีมใช้ร่วมกัน และโค้ดส่วนกลางที่ 3 คนแบ่งกันทำคนละชิ้นในชั่วโมงแรก (เจ้าของแต่ละชิ้นดูที่โครงไฟล์และเจ้าของงาน) จากนั้นเปิดไฟล์ของตัวเอง: คนที่ 1 งานบุคลากร · คนที่ 2 ผู้รับการประเมิน · คนที่ 3 กรรมการ

                              วิธีอ่านเอกสารนี้

                              แต่ละเส้น API แบ่งเป็น 4 ส่วน เรียงตามทางที่ข้อมูลเดินทาง

                              1. Frontend: Zod ฝั่งหน้าเว็บ, ฟอร์ม, โค้ดเรียก API และการแสดงผล
                              2. Backend: เส้นทาง URL, การตรวจสิทธิ์, Zod และ route
                              3. Database: ตาราง คอลัมน์ และคำสั่ง SQL
                              4. วิธีทดสอบ: สิ่งที่ส่งไป และผลที่ได้จริง
                              • เลขบรรทัดในกล่องโค้ดตรงกับไฟล์จริง กดปุ่ม 📍 เพื่อคัดลอกตำแหน่ง แล้วกด Ctrl + P ใน VS Code วางแล้วกด Enter จะเปิดไฟล์ตรงบรรทัดนั้น
                              • โค้ดในเอกสารตรงกับไฟล์จริง ณ วันที่ 10 พฤศจิกายน 2569 ถ้าแก้โค้ดภายหลัง เลขบรรทัดอาจคลาดได้ (ครูสร้างใหม่ได้ด้วย npm run map ใน docs/tools)
                              • ผลทดสอบมาจากการยิง API จริงบนฐานข้อมูลว่าง ถ้าผลเป็นรายการยาวจะแสดงเพียงรายการแรก และข้อความยาวจะถูกตัดด้วย ...
                              • โค้ดที่ทุกเส้นใช้ร่วมกันอยู่ที่ส่วนกลาง เส้นอื่นจะไม่เขียนซ้ำ

                              เริ่มต้นโปรเจกต์ (ไม่มีข้อมูลตัวอย่าง)

                              ระบบนี้ไม่ใช้ seed หรือ mock data ทุกคนใช้ข้อมูลจริงที่สร้างผ่านหน้าเว็บ

                              1. รัน docker compose up -d mysql จะได้ตารางเปล่า
                              2. สมัครสมาชิกที่หน้า /register ตามเส้นสมัครสมาชิก
                              3. เปลี่ยนบัญชีแรกเป็น hr ในฐานข้อมูล แล้ว login ใหม่
                              4. hr สร้างกรรมการ หัวข้อ ตัวชี้วัด และมอบหมายงานผ่านหน้าเว็บ

                              ใช้ฐานข้อมูลร่วมกันผ่านวง LAN

                              • ให้เครื่องของคนที่ 2 เปิด MySQL เพราะฟังก์ชัน Backup เรียก docker exec ไปยัง container บนเครื่องเดียวกัน
                              • เครื่องอื่นแก้ DB_HOST ใน .env เป็น IP ของเครื่องนั้น และเครื่องคนที่ 2 ต้องเปิด port 3306 ใน Windows Firewall
                              • ไฟล์หลักฐานอยู่ใน backend/uploads/ ของเครื่องที่รับอัปโหลด จึงต้องทดสอบการดาวน์โหลดไฟล์บน Backend ตัวเดียวกัน
                              • ก่อนทดสอบ Restore ต้องบอกเพื่อน เพราะข้อมูลของทุกคนจะถูกเขียนทับ
                              • ล็อก schema.sql ให้จบตั้งแต่ต้น ไฟล์นี้รันแค่ตอนสร้างฐานข้อมูลครั้งแรก ถ้าแก้ภายหลังต้อง docker compose down -v ซึ่งลบข้อมูลทั้งหมด

export const validate = (schema) => (req, res, next) => {
  const result = schema.safeParse(req.body ?? {});
  if (!result.success) {
    const errors = result.error.issues.map((i) => ({ field: i.path.join("."), message: i.message }));
    return fail(res, 400, errors[0].message, errors);
  }
  req.body = result.data; // ข้อมูลที่ผ่านการ trim/แปลงแล้ว
  next();
};

                              5. ตรวจ token · คนที่ 1 ทำ

                              อ่านจาก header Authorization: Bearer ... ถ้าถูกต้องจะได้ req.user คือ { id, role, full_name }

                              js
                              // ตรวจ JWT จาก header: Authorization: Bearer <token>
export function verifyToken(req, res, next) {
  const header = req.headers.authorization || "";
  const token = header.startsWith("Bearer ") ? header.slice(7) : null;
  if (!token) return fail(res, 401, "กรุณาเข้าสู่ระบบ");
  try {
    req.user = jwt.verify(token, process.env.JWT_SECRET); // { id, role, full_name }
    next();
  } catch {
    fail(res, 401, "Token ไม่ถูกต้องหรือหมดอายุ");
  }
}

export const validateWith = (schema) => (values) => {
  const result = schema.safeParse(values);
  if (result.success) return { values: result.data, errors: {} };

  const errors = {};
  for (const issue of result.error.issues) {

  return { values: {}, errors };

export function useAuth(roles) {
  const router = useRouter();
  const [user, setUser] = useState(null);

  useEffect(() => {
    const u = getUser();
    if (!u) {
      router.replace("/login");
    } else if (u.must_change_password && router.pathname !== "/change-password") {
      router.replace("/change-password"); // บังคับเปลี่ยนรหัสผ่านครั้งแรก

    } else {

      setUser(u); // eslint-disable-line react-hooks/set-state-in-effect
    }
  }, []); // eslint-disable-line react-hooks/exhaustive-deps

  return user;
}

// GET /api/topics  หัวข้อทั้งหมด พร้อมตัวชี้วัดของแต่ละหัวข้อ (ทุก role ดูได้)

  const [topics] = await pool.query(`SELECT ${TOPIC_FIELDS} FROM evaluation_topics ORDER BY id`);
  const [indicators] = await pool.query("SELECT * FROM indicators ORDER BY id");
  for (const t of topics) t.indicators = indicators.filter((i) => i.topic_id === t.id);
  ok(res, topics);
});

// GET /api/topics/:id
router.get("/:id", requireRole("hr", "evaluatee", "committee"), async (req, res) => {
  const [rows] = await pool.query(`SELECT ${TOPIC_FIELDS} FROM evaluation_topics WHERE id = ?`, [req.params.id]);
  if (!rows[0]) return fail(res, 404, "ไม่พบหัวข้อการประเมิน");
  const [indicators] = await pool.query("SELECT * FROM indicators WHERE topic_id = ? ORDER BY id", [req.params.id]);
  ok(res, { ...rows[0], indicators });
});

// POST /api/topics  body: { title, description, start_date, end_date }
router.post("/", requireRole("hr"), validate(topicSchema), async (req, res) => {

  const [result] = await pool.query(
    "INSERT INTO evaluation_topics (title, description, start_date, end_date) VALUES (?, ?, ?, ?)",
    [title, description, start_date, end_date]

                              อาการ สาเหตุ วิธีแก้
                              401 กรุณาเข้าสู่ระบบ ไม่ได้ส่ง header Authorization ใส่ Bearer <token>
                              401 Token ไม่ถูกต้องหรือหมดอายุ token เกิน 8 ชั่วโมง หรือ JWT_SECRET เปลี่ยน login ใหม่

                              6. hash รหัสผ่าน

                              สุ่ม salt แล้วเข้ารหัสด้วย crypto.scrypt เก็บเป็น salt:hash ใน DB ไม่มีรหัสผ่านจริงเก็บอยู่เลย

                              js
                              // เก็บในรูป "salt:hash" (hex)
export async function hashPassword(password) {
  const salt = crypto.randomBytes(16).toString("hex");
  const hash = await scrypt(password, salt, 64);
  return `${salt}:${hash.toString("hex")}`;
}

router.post("/login", validate(loginSchema), async (req, res) => {
  const { username, password } = req.body;
  const [rows] = await pool.query("SELECT * FROM users WHERE username = ?", [username]);
  const user = rows[0];
  if (!user || !(await verifyPassword(password, user.password_hash))) {
    return fail(res, 401, "ชื่อผู้ใช้หรือรหัสผ่านไม่ถูกต้อง");

  const token = jwt.sign(
    { id: user.id, role: user.role, full_name: user.full_name },
    process.env.JWT_SECRET,
    { expiresIn: process.env.JWT_EXPIRES_IN || "8h" }
  );
  ok(res, {
    token,
    user: {
      id: user.id,
      username: user.username,
      full_name: user.full_name,
      role: user.role,
      must_change_password: !!user.must_change_password,
    },
  }, "เข้าสู่ระบบสำเร็จ");
});

                              export async function verifyPassword(password, stored) {
  const [salt, hash] = stored.split(":");
  const test = await scrypt(password, salt, 64);
  return crypto.timingSafeEqual(Buffer.from(hash, "hex"), test);
}

router.put("/change-password", verifyToken, requireRole(...ALL_ROLES), validate(changePasswordSchema), async (req, res) => {
  const { old_password, new_password } = req.body;
  const [rows] = await pool.query("SELECT password_hash FROM users WHERE id = ?", [req.user.id]);
  if (!rows[0]) return fail(res, 404, "ไม่พบผู้ใช้");
  if (!(await verifyPassword(old_password, rows[0].password_hash))) {
    return fail(res, 400, "รหัสผ่านเดิมไม่ถูกต้อง");
  }
  if (old_password === new_password) return fail(res, 400, "รหัสผ่านใหม่ต้องไม่ซ้ำรหัสผ่านเดิม");
  await pool.query("UPDATE users SET password_hash = ?, must_change_password = FALSE WHERE id = ?", [
    await hashPassword(new_password),
    req.user.id,
  ]);
  ok(res, null, "เปลี่ยนรหัสผ่านสำเร็จ");
});

export async function hashPassword(password) {
  const salt = crypto.randomBytes(16).toString("hex");
  const hash = await scrypt(password, salt, 64);
  return `${salt}:${hash.toString("hex")}`;
}

export async function verifyPassword(password, stored) {
  const [salt, hash] = stored.split(":");
  const test = await scrypt(password, salt, 64);
  return crypto.timingSafeEqual(Buffer.from(hash, "hex"), test);
}

// ตรวจ JWT จาก header: Authorization: Bearer <token>
export function verifyToken(req, res, next) {
  const header = req.headers.authorization || "";
  const token = header.startsWith("Bearer ") ? header.slice(7) : null;
  if (!token) return fail(res, 401, "กรุณาเข้าสู่ระบบ");
  try {
    req.user = jwt.verify(token, process.env.JWT_SECRET); // { id, role, full_name }
    next();
  } catch {
    fail(res, 401, "Token ไม่ถูกต้องหรือหมดอายุ");
  }
}

                              // ฟังก์ชันคำนวณคะแนน (ไม่มี DB ทดสอบด้วย node --test ได้)

export const round2 = (n) => Math.round(n * 100) / 100;

export const sum = (nums) => round2(nums.reduce((a, b) => a + b, 0));

// ค่าเฉลี่ย ไม่นับค่าว่าง (null/undefined) ถ้าไม่มีเลยคืน null
export function average(nums) {
  const values = nums.filter((n) => n != null);
  return values.length ? round2(sum(values) / values.length) : null;
}

export const percent = (part, whole) => (whole ? round2((part / whole) * 100) : 0);

// คะแนนถ่วงน้ำหนัก: มี/ไม่มี -> 0 หรือ 1 เท่าของน้ำหนัก, สเกล 1-4 -> (คะแนน/4) x น้ำหนัก
export function weighted(score, indicator) {
  if (score == null) return 0;
  const ratio = indicator.score_type === "yes_no" ? score : score / 4;
  return round2(ratio * indicator.weight);
}

// คะแนนถูกต้องตามรูปแบบหรือไม่: มี/ไม่มี = 0 หรือ 1, สเกล = 1-4
export function isValidScore(score, scoreType) {
  if (!Number.isInteger(score)) return false;
  return scoreType === "yes_no" ? score === 0 || score === 1 : score >= 1 && score <= 4;
}

// ความคืบหน้าของตัวชี้วัด 1 ตัว (%): กรอกคะแนนแล้ว + แนบหลักฐานแล้ว (ถ้าตัวชี้วัดนั้นต้องแนบ)
export function indicatorProgress(indicator, score, evidenceCount) {
  const steps = [score != null];
  if (indicator.evidence_type !== "none") steps.push(evidenceCount > 0);
  return percent(steps.filter(Boolean).length, steps.length);
}

// เรียงข้อมูลตาม key (ค่าว่างไว้ท้ายเสมอ) ไม่แก้ array เดิม
export function sortBy(list, key, dir = "asc") {
  const sign = dir === "desc" ? -1 : 1;
  return [...list].sort((a, b) => {
    if (a[key] == null) return 1;
    if (b[key] == null) return -1;
    return a[key] > b[key] ? sign : a[key] < b[key] ? -sign : 0;
  });
}

// สรุปผลของผู้รับประเมิน 1 คน
// indicators: ตัวชี้วัดทั้งหมด, selfData: { indicator_id: { score, evidence_count } }
// assignments: กรรมการของคนนี้, committeeScores: [{ assignment_id, indicator_id, score }]
// ผลกรรมการใช้ค่าเฉลี่ยจากกรรมการที่ "ยืนยันแล้ว" เท่านั้น
export function buildResult(indicators, selfData, assignments, committeeScores) {
  const confirmedIds = assignments.filter((a) => a.status === "confirmed").map((a) => a.id);

  const rows = indicators.map((ind) => {
    const scores = {}; // { assignment_id: score }
    for (const s of committeeScores) if (s.indicator_id === ind.id) scores[s.assignment_id] = s.score;
    const self_score = selfData[ind.id]?.score ?? null;
    const evidence_count = selfData[ind.id]?.evidence_count ?? 0;
    const avg_score = average(confirmedIds.map((id) => scores[id]));
    return {
      ...ind,
      self_score,
      evidence_count,
      progress: indicatorProgress(ind, self_score, evidence_count),
      scores,
      avg_score,
      self_weighted: weighted(self_score, ind),
      committee_weighted: weighted(avg_score, ind),
    };
  });

  const total_weight = sum(rows.map((r) => r.weight));
  const filled = rows.filter((r) => r.progress === 100).length; // ตัวชี้วัดที่บันทึกครบ
  const committee_total = sum(rows.map((r) => r.committee_weighted));

  return {
    rows,
    summary: {
      total_weight,
      indicator_count: rows.length,
      filled,
      progress: percent(filled, rows.length),
      self_total: sum(rows.map((r) => r.self_weighted)),
      committee_count: assignments.length,
      confirmed_count: confirmedIds.length,
      committee_total: confirmedIds.length ? committee_total : null,
      committee_percent: confirmedIds.length ? percent(committee_total, total_weight) : null,
    },
  };
}
