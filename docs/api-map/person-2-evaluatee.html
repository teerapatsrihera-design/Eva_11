        GET /api/profile ข้อมูลส่วนตัวของตัวเอง ทุกบทบาทที่ login แล้ว ต้อง login แล้ว
        PUT /api/profile แก้ไขข้อมูลส่วนตัว ทุกบทบาทที่ login แล้ว ต้อง login แล้ว
        GET /api/self-evaluations แบบประเมินตนเอง พร้อมคะแนนและความคืบหน้า ผู้รับการประเมิน ควรมีหัวข้อและตัวชี้วัดจากคนที่ 1 ถ้ายังไม่มีจะได้ topics: []
        PUT /api/self-evaluations/{indicatorId} กรอกหรือแก้คะแนนตนเอง ผู้รับการประเมิน ต้องมีตัวชี้วัดจากคนที่ 1 และหัวข้ออยู่ในช่วงเปิด
        POST /api/self-evaluations/{indicatorId}/evidences เพิ่มรายละเอียดประกอบและอัปโหลดไฟล์หรือ URL ผู้รับการประเมิน ต้องมีตัวชี้วัดจากคนที่ 1 และหัวข้ออยู่ในช่วงเปิด
        GET /api/evidences/{id}/file ดาวน์โหลดไฟล์หลักฐาน ทุกบทบาทที่ login แล้ว ต้องมีหลักฐานที่แนบไฟล์
        DELETE /api/evidences/{id} ลบรายละเอียดประกอบ ผู้รับการประเมิน ต้องมีหลักฐานจากเส้นเพิ่มหลักฐาน
        GET /api/backups รายการไฟล์สำรอง hr บัญชี hr
        POST /api/backups สำรองข้อมูลตอนนี้ hr บัญชี hr และต้องตั้ง DB_CONTAINER ให้ตรงชื่อ container ของ MySQL
        GET /api/backups/{name}/download ดาวน์โหลดไฟล์สำรอง hr ต้องมีไฟล์จาก POST /api/backups
        POST /api/backups/{name}/restore กู้คืนจากไฟล์สำรองบน server hr ต้องมีไฟล์จาก POST /api/backups
        POST /api/backups/restore กู้คืนจากไฟล์ .sql ที่อัปโหลด hr ต้องมีไฟล์ .sql ที่ดาวน์โหลดไว้
        DELETE /api/backups/{name} ลบไฟล์สำรอง hr ต้องมีไฟล์จาก POST /api/backups
        GET /api/reports/overview ข้อมูล Dashboard hr ทำได้ทันที แต่จะเห็นตัวเลขจริงเมื่อมีผู้ใช้ ตัวชี้วัด และคะแนนแล้ว

        js
        // GET /api/self-evaluations  หัวข้อ > ตัวชี้วัด พร้อมคะแนนตนเอง รายละเอียด และความคืบหน้า
router.get("/", async (req, res) => {
  const userId = req.user.id;
  const [topics] = await pool.query(
    "SELECT *, (CURDATE() BETWEEN start_date AND end_date) AS is_open FROM evaluation_topics ORDER BY id"
  );
  const [indicators] = await pool.query("SELECT * FROM indicators ORDER BY id");
  const [selfRows] = await pool.query("SELECT * FROM self_evaluations WHERE evaluatee_id = ?", [userId]);
  const [evidences] = await pool.query(
    `SELECT ev.* FROM evidences ev JOIN self_evaluations se ON se.id = ev.self_evaluation_id
     WHERE se.evaluatee_id = ? ORDER BY ev.id`,
    [userId]
  );

  let filled = 0;
  for (const t of topics) {
    t.indicators = indicators
      .filter((i) => i.topic_id === t.id)
      .map((i) => {
        const self = selfRows.find((s) => s.indicator_id === i.id);
        const list = self ? evidences.filter((e) => e.self_evaluation_id === self.id) : [];
        const score = self?.score ?? null;
        const progress = indicatorProgress(i, score, list.length);
        if (progress === 100) filled++;
        return { ...i, score, evidences: list, progress };
      });

  const [rows] = await pool.query(
    `SELECT ev.*, se.evaluatee_id, (CURDATE() BETWEEN t.start_date AND t.end_date) AS is_open
     FROM evidences ev
     JOIN self_evaluations se ON se.id = ev.self_evaluation_id

        3. เรียกโปรแกรม mysqldump/mysql

        DB_CONTAINER มีค่าจะเรียกผ่าน docker exec ไม่มีค่าจะเรียกตรง และส่งรหัสผ่านทาง MYSQL_PWD ไม่ให้อยู่บน command line

        js
        // เรียก mysqldump / mysql แล้วรอจนจบ: input = ข้อมูลที่ส่งเข้า stdin, คืน stdout เป็น Buffer
// - รันบนเครื่อง (npm run dev, มี DB_CONTAINER): เรียกผ่าน docker exec ใน container ของ MySQL
// - รันใน Docker (docker compose): image ของ backend ติดตั้ง mysql client ไว้ เรียกตรงไปที่ DB_HOST
// ส่งรหัสผ่านทาง MYSQL_PWD เพื่อไม่ให้รหัสผ่านอยู่บน command line
function runMysql(tool, options, input) {
  const { DB_CONTAINER, DB_HOST, DB_PORT, DB_USER, DB_PASSWORD, DB_NAME } = process.env;
  const args = [`-u${DB_USER}`, "--default-character-set=utf8mb4", ...options, DB_NAME];
  const [cmd, cmdArgs] = DB_CONTAINER
    ? ["docker", ["exec", "-i", "-e", `MYSQL_PWD=${DB_PASSWORD}`, DB_CONTAINER, tool, ...args]]
    : [tool, [`-h${DB_HOST}`, `-P${DB_PORT}`, "--skip-ssl", ...args]];

  return new Promise((resolve, reject) => {
    const child = spawn(cmd, cmdArgs, { env: { ...process.env, MYSQL_PWD: DB_PASSWORD } });
    const output = [];
    let error = "";
    child.stdout.on("data", (chunk) => output.push(chunk));
    child.stderr.on("data", (chunk) => (error += chunk));
    child.on("error", reject); // เช่น เครื่องไม่มีคำสั่ง docker / mysqldump
    child.on("close", (code) => (code === 0 ? resolve(Buffer.concat(output)) : reject(new Error(error.trim()))));
    child.stdin.end(input);
  });
}

        4. คำนวณผลรายคน

        ดึงข้อมูลทีเดียวทุกคนด้วย IN (?) แล้วแยกตามคน ลดจำนวนครั้งที่ถาม DB

        js
        // คำนวณผลของผู้รับประเมินหลายคนในครั้งเดียว คืน { evaluatee_id: { assignments, rows, summary } }
async function loadResults(evaluateeIds) {
  if (!evaluateeIds.length) return {};
  const indicators = await loadIndicators();
  // คะแนนตนเอง + จำนวนหลักฐาน
  const [selfRows] = await pool.query(
    `SELECT se.evaluatee_id, se.indicator_id, se.score, COUNT(ev.id) AS evidence_count
     FROM self_evaluations se LEFT JOIN evidences ev ON ev.self_evaluation_id = se.id
     WHERE se.evaluatee_id IN (?) GROUP BY se.id`,
    [evaluateeIds]
  );
  const [assignments] = await pool.query(
    `SELECT a.*, u.full_name AS committee_name, u.position AS committee_position
     FROM assignments a JOIN users u ON u.id = a.committee_id
     WHERE a.evaluatee_id IN (?) ORDER BY a.committee_role, a.id`,
    [evaluateeIds]
  );
  const [scores] = await pool.query(
    `SELECT s.assignment_id, s.indicator_id, s.score
     FROM committee_scores s JOIN assignments a ON a.id = s.assignment_id
     WHERE a.evaluatee_id IN (?)`,
    [evaluateeIds]
  );

  const results = {};
  for (const id of evaluateeIds) {
    const mine = assignments.filter((a) => a.evaluatee_id === id);
    const selfData = Object.fromEntries(selfRows.filter((r) => r.evaluatee_id === id).map((r) => [r.indicator_id, r]));
    const myScores = scores.filter((s) => mine.some((a) => a.id === s.assignment_id));
    results[id] = { assignments: mine, ...buildResult(indicators, selfData, mine, myScores) };
  }
  return results;
}

        5. สรุปผล 1 คน

        ใช้คะแนนเฉลี่ยจากกรรมการที่ยืนยันแล้วเท่านั้น

        js
        // สรุปผลของผู้รับประเมิน 1 คน
// indicators: ตัวชี้วัดทั้งหมด, selfData: { indicator_id: { score, evidence_count } }
// assignments: กรรมการของคนนี้, committeeScores: [{ assignment_id, indicator_id, score }]
// ผลกรรมการใช้ค่าเฉลี่ยจากกรรมการที่ "ยืนยันแล้ว" เท่านั้น
export function buildResult(indicators, selfData, assignments, committeeScores) {
  const confirmedIds = assignments.filter((a) => a.status === "confirmed").map((a) => a.id);

  const rows = indicators.map((ind) => {
    const scores = {}; // { assignment_id: score }
    for (const s of committeeScores) if (s.indicator_id === ind.id) scores[s.assignment_id] = s.score;
    const self_score = selfData[ind.id]?.score ?? null;
    const evidence_count = selfData[ind.id]?.evidence_count ?? 0;
    const avg_score = average(confirmedIds.map((id) => scores[id]));
    return {
      ...ind,
      self_score,
      evidence_count,
      progress: indicatorProgress(ind, self_score, evidence_count),
      scores,
      avg_score,
      self_weighted: weighted(self_score, ind),
      committee_weighted: weighted(avg_score, ind),
    };
  });

  const total_weight = sum(rows.map((r) => r.weight));
  const filled = rows.filter((r) => r.progress === 100).length; // ตัวชี้วัดที่บันทึกครบ
  const committee_total = sum(rows.map((r) => r.committee_weighted));

  return {
    rows,
    summary: {
      total_weight,
      indicator_count: rows.length,
      filled,
      progress: percent(filled, rows.length),
      self_total: sum(rows.map((r) => r.self_weighted)),
      committee_count: assignments.length,
      confirmed_count: confirmedIds.length,
      committee_total: confirmedIds.length ? committee_total : null,
      committee_percent: confirmedIds.length ? percent(committee_total, total_weight) : null,
    },
  };
}

export function average(nums) {
  const values = nums.filter((n) => n != null);
  return values.length ? round2(sum(values) / values.length) : null;
}

        // เรียงข้อมูลตาม key (ค่าว่างไว้ท้ายเสมอ) ไม่แก้ array เดิม
export function sortBy(list, key, dir = "asc") {
  const sign = dir === "desc" ? -1 : 1;
  return [...list].sort((a, b) => {
    if (a[key] == null) return 1;
    if (b[key] == null) return -1;
    return a[key] > b[key] ? sign : a[key] < b[key] ? -sign : 0;
  });
}

        8. ตรวจความครบถ้วน

        เช่น น้ำหนักรวมต้องเป็น 100 ทุกคนต้องมีประธานกรรมการ ตามเกณฑ์ 8.2

        js
        // ตรวจความถูกต้อง/ครบถ้วนของข้อมูลการประเมิน คืนรายการข้อความแจ้งเตือน (ว่าง = ข้อมูลพร้อม)
// topics: [{ id, title }], indicators: [{ topic_id, weight }]
// evaluatees: [{ id, full_name }], assignments: [{ evaluatee_id, committee_role }]
export function findDataIssues({ topics, indicators, evaluatees, assignments }) {
  const issues = [];

  const totalWeight = sum(indicators.map((i) => i.weight));
  if (totalWeight !== 100) issues.push(`น้ำหนักรวมของตัวชี้วัดทั้งหมดเท่ากับ ${totalWeight} (ควรเท่ากับ 100)`);

  for (const t of topics) {
    if (!indicators.some((i) => i.topic_id === t.id)) issues.push(`หัวข้อ "${t.title}" ยังไม่มีตัวชี้วัด`);
  }

  for (const e of evaluatees) {
    const mine = assignments.filter((a) => a.evaluatee_id === e.id);
    if (mine.length === 0) issues.push(`${e.full_name} ยังไม่มีกรรมการประเมิน`);
    else if (!mine.some((a) => a.committee_role === "chair")) issues.push(`${e.full_name} ยังไม่มีประธานกรรมการ`);
  }
  return issues;
}

// เรียก mysqldump / mysql แล้วรอจนจบ: input = ข้อมูลที่ส่งเข้า stdin, คืน stdout เป็น Buffer
// - รันบนเครื่อง (npm run dev, มี DB_CONTAINER): เรียกผ่าน docker exec ใน container ของ MySQL
// - รันใน Docker (docker compose): image ของ backend ติดตั้ง mysql client ไว้ เรียกตรงไปที่ DB_HOST
// ส่งรหัสผ่านทาง MYSQL_PWD เพื่อไม่ให้รหัสผ่านอยู่บน command line
function runMysql(tool, options, input) {
  const { DB_CONTAINER, DB_HOST, DB_PORT, DB_USER, DB_PASSWORD, DB_NAME } = process.env;
  const args = [`-u${DB_USER}`, "--default-character-set=utf8mb4", ...options, DB_NAME];
  const [cmd, cmdArgs] = DB_CONTAINER
    ? ["docker", ["exec", "-i", "-e", `MYSQL_PWD=${DB_PASSWORD}`, DB_CONTAINER, tool, ...args]]
    : [tool, [`-h${DB_HOST}`, `-P${DB_PORT}`, "--skip-ssl", ...args]];

  return new Promise((resolve, reject) => {
    const child = spawn(cmd, cmdArgs, { env: { ...process.env, MYSQL_PWD: DB_PASSWORD } });
    const output = [];
    let error = "";
    child.stdout.on("data", (chunk) => output.push(chunk));
    child.stderr.on("data", (chunk) => (error += chunk));
    child.on("error", reject); // เช่น เครื่องไม่มีคำสั่ง docker / mysqldump
    child.on("close", (code) => (code === 0 ? resolve(Buffer.concat(output)) : reject(new Error(error.trim()))));
    child.stdin.end(input);
  });
}

// สำรองฐานข้อมูลเป็นไฟล์ .sql คืนชื่อไฟล์ เช่น backup-2026-11-07-10-30-00.sql

  const stamp = new Date().toLocaleString("sv-SE").replace(/[: ]/g, "-"); // 2026-11-07 10:30:00 -> 2026-11-07-10-30-00
  const name = `backup-${stamp}${suffix}.sql`;

  fs.writeFileSync(path.join(BACKUP_DIR, name), sql);
  return name;
}

  return fs
    .readdirSync(BACKUP_DIR)
    .filter(isBackupName)
    .map((name) => {
      const stat = fs.statSync(path.join(BACKUP_DIR, name));
      return { name, size: stat.size, created_at: stat.mtime };
    })
    .sort((a, b) => b.created_at - a.created_at);
}

        import { useEffect, useRef, useState } from "react";
import { DatabaseBackup, Download, RotateCcw, Trash2, Upload } from "lucide-react";
import api, { errorMessage, downloadFile } from "@/lib/api";
import { useAuth } from "@/lib/auth";
import Layout from "@/components/Layout";

// สำรองและกู้คืนข้อมูล (mysqldump ผ่าน backend)
export default function Backups() {
  const user = useAuth(["hr"]);
  const [list, setList] = useState([]);
  const [busy, setBusy] = useState(false);
  const [message, setMessage] = useState({ type: "", text: "" });
  const fileRef = useRef(null);

  const load = () =>
    api
      .get("/backups")
      .then((res) => setList(res.data.data))
      .catch((err) => setMessage({ type: "error", text: errorMessage(err) }));

  useEffect(() => {
    if (user) load();
  }, [user]);

  // เรียก API ที่ใช้เวลานาน: ล็อกปุ่ม แสดงผลลัพธ์ แล้วโหลดรายการใหม่
  const run = async (request) => {
    setBusy(true);
    setMessage({ type: "", text: "" });
    try {
      const res = await request();
      setMessage({ type: "success", text: res.data.message });
      load();
    } catch (err) {
      setMessage({ type: "error", text: errorMessage(err) });
    } finally {
      setBusy(false);
    }
  };

  const restoreFrom = (b) =>
    confirm(`กู้คืนข้อมูลจาก ${b.name} ?\nข้อมูลปัจจุบันจะถูกแทนที่ (ระบบจะสำรองข้อมูลปัจจุบันไว้ให้ก่อน)`) &&
    run(() => api.post(`/backups/${b.name}/restore`));

  const restoreUpload = () => {
    const file = fileRef.current.files[0];
    if (!file) return setMessage({ type: "error", text: "กรุณาเลือกไฟล์ .sql" });
    if (!confirm(`กู้คืนข้อมูลจากไฟล์ ${file.name} ?\nข้อมูลปัจจุบันจะถูกแทนที่`)) return;
    const form = new FormData();
    form.append("file", file);
    run(() => api.post("/backups/restore", form)).then(() => (fileRef.current.value = ""));
  };

  const remove = (b) => confirm(`ลบไฟล์ ${b.name} ?`) && run(() => api.delete(`/backups/${b.name}`));

  if (!user) return null;

  return (
    <Layout user={user}>
      <div className="flex flex-wrap justify-between items-center gap-2 mb-4">
        <h1 className="text-2xl font-bold">สำรองและกู้คืนข้อมูล</h1>
        <button className="btn btn-primary" disabled={busy} onClick={() => run(() => api.post("/backups"))}>
          <DatabaseBackup size={18} /> สำรองข้อมูลตอนนี้
        </button>
      </div>

      {busy && <progress className="progress progress-primary w-full mb-2" />}
      {message.text && <div className={`alert alert-${message.type} mb-4`}>{message.text}</div>}

      <div className="card bg-base-100 shadow mb-4">
        <div className="card-body py-4 flex-row flex-wrap items-center gap-2">
          <span className="font-medium">กู้คืนจากไฟล์ในเครื่อง</span>
          <input ref={fileRef} type="file" accept=".sql" className="file-input file-input-sm w-full sm:w-auto" />
          <button className="btn btn-sm btn-warning" disabled={busy} onClick={restoreUpload}>
            <Upload size={16} /> อัปโหลดและกู้คืน
          </button>
        </div>
      </div>

      <div className="overflow-x-auto bg-base-100 rounded-box shadow">
        <table className="table">
          <thead>
            <tr>
              <th>ไฟล์สำรอง</th>
              <th>วันเวลา</th>
              <th className="text-right">ขนาด</th>
              <th></th>
            </tr>
          </thead>
          <tbody>
            {list.map((b) => (
              <tr key={b.name}>
                <td className="break-all">{b.name}</td>
                <td className="whitespace-nowrap">{new Date(b.created_at).toLocaleString("th-TH")}</td>
                <td className="text-right whitespace-nowrap">{(b.size / 1024).toFixed(1)} KB</td>
                <td className="whitespace-nowrap text-right">
                  <button
                    className="btn btn-xs btn-ghost"
                    title="ดาวน์โหลด"
                    onClick={() => downloadFile(`/backups/${b.name}/download`, b.name).catch((err) => alert(errorMessage(err)))}
                  >
                    <Download size={14} />
                  </button>
                  <button className="btn btn-xs btn-ghost" title="กู้คืน" disabled={busy} onClick={() => restoreFrom(b)}>
                    <RotateCcw size={14} />
                  </button>
                  <button className="btn btn-xs btn-ghost text-error" title="ลบ" disabled={busy} onClick={() => remove(b)}>
                    <Trash2 size={14} />
                  </button>
                </td>
              </tr>
            ))}
          </tbody>
        </table>
        {list.length === 0 && <p className="text-center opacity-60 p-4">ยังไม่มีไฟล์สำรอง</p>}
      </div>
    </Layout>
  );
