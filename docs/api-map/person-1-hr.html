        งานบุคลากร (hr)

        API ของงานบุคลากร: ผู้ใช้ หัวข้อ ตัวชี้วัด และการมอบหมายกรรมการ

        เริ่มต้นอ่าน

        แผนที่โค้ดรายเส้น API ของคนที่รับผิดชอบส่วนนี้ มี 21 เส้น

        อ่านก่อนเริ่ม: ไฟล์ส่วนกลาง (ลำดับงาน, โค้ดส่วนกลาง, แม่แบบ CRUD และปัญหาที่เจอบ่อย)

        เกณฑ์หลัก: 5.1.1-5.1.9 งานบุคลากร (หัวข้อ ตัวชี้วัด ผู้ใช้ มอบหมายกรรมการ)

        ส่วนกลางที่คนที่ 1 ทำในชั่วโมงแรก: สร้างโปรเจกต์, backend ส่วนกลาง และ Field, Modal, Layout, StatusBadge (ดูโค้ดในไฟล์ส่วนกลาง)

        4. route รายชื่อผู้ใช้

        เริ่มด้วย WHERE 1 = 1 เพื่อให้ต่อเงื่อนไข AND ได้ง่าย และใช้ ? แทนการต่อข้อความ เพื่อกัน SQL Injection

        js
        // GET /api/users?role=evaluatee&q=สมชาย
router.get("/", async (req, res) => {
  const { role, q } = req.query;
  let sql = `SELECT ${USER_FIELDS} FROM users WHERE 1 = 1`;
  const params = [];
  if (role) {
    sql += " AND role = ?";
    params.push(role);
  }
  if (q) {
    sql += " AND (full_name LIKE ? OR username LIKE ?)";
    params.push(`%${q}%`, `%${q}%`);
  }
  const [rows] = await pool.query(sql + " ORDER BY id", params);
  ok(res, rows);
});

        3. Database

        โครงตาราง users

        sql
        CREATE TABLE users (
  id INT AUTO_INCREMENT PRIMARY KEY,
  username VARCHAR(50) NOT NULL UNIQUE,
  password_hash VARCHAR(255) NOT NULL,
  full_name VARCHAR(100) NOT NULL,
  email VARCHAR(100) UNIQUE,
  phone VARCHAR(20),
  position VARCHAR(100),
  department VARCHAR(100),
  role ENUM('hr', 'evaluatee', 'committee') NOT NULL,
  must_change_password BOOLEAN NOT NULL DEFAULT TRUE,
  created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  updated_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
);

        คำสั่ง SQL ที่เส้นนี้ใช้

        sql
        SELECT ${USER_FIELDS} FROM users WHERE 1 = 1
-- ต่อท้ายเมื่อมีตัวกรอง
AND role = ?
-- ต่อท้ายเมื่อมีตัวกรอง
AND (full_name LIKE ? OR username LIKE ?)
-- ต่อท้ายเสมอ
ORDER BY id;

router.post("/import", async (req, res) => {
  const file = req.files?.file;
  if (!file) return fail(res, 400, "กรุณาเลือกไฟล์ CSV");
  if (!file.name.toLowerCase().endsWith(".csv")) return fail(res, 400, "รองรับเฉพาะไฟล์ .csv");

  // แยกบรรทัด ตัด BOM และบรรทัดว่าง
  const lines = file.data.toString("utf8").replace(/^﻿/, "").split(/\r?\n/).filter((l) => l.trim());
  const headers = lines[0]?.split(",").map((h) => h.trim());
  if (!headers?.includes("username") || lines.length < 2) {
    return fail(res, 400, "ไฟล์ต้องมีหัวตารางและข้อมูลอย่างน้อย 1 แถว");
  }

  // ตรวจทุกแถวก่อน ถ้ามีแถวผิดไม่บันทึกเลย
  const users = [];
  const errors = [];
  lines.slice(1).forEach((line, i) => {
    const values = line.split(",").map((v) => v.trim());
    const row = Object.fromEntries(headers.map((h, j) => [h, values[j] ?? ""]));
    const result = userSchema.safeParse(row);
    if (result.success) users.push(result.data);
    else errors.push({ row: i + 2, message: result.error.issues[0].message });
  });
  if (errors.length) return fail(res, 400, `ข้อมูลไม่ถูกต้อง ${errors.length} แถว`, errors);

  // INSERT ทีเดียวทุกแถว (ถ้าซ้ำจะ error ทั้งก้อน ไม่มีแถวไหนถูกบันทึก)
  const created = [];
  const values = [];
  for (const u of users) {
    const password = tempPassword();
    created.push({ username: u.username, full_name: u.full_name, temp_password: password });
    values.push([u.username, await hashPassword(password), u.full_name, u.email, u.phone, u.position, u.department, u.role]);
  }
  await pool.query(
    "INSERT INTO users (username, password_hash, full_name, email, phone, position, department, role) VALUES ?",
    [values]
  );
  ok(res, created, `นำเข้าผู้ใช้สำเร็จ ${created.length} คน`, 201);
});

        js
        // PUT /api/indicators/:id
router.put("/:id", requireRole("hr"), validate(indicatorSchema), async (req, res) => {
  const i = req.body;
  const [[old]] = await pool.query("SELECT score_type FROM indicators WHERE id = ?", [req.params.id]);
  if (!old) return fail(res, 404, "ไม่พบตัวชี้วัด");

  // มีคะแนนแล้ว ห้ามเปลี่ยนรูปแบบการประเมิน (คะแนนเดิมจะผิดความหมาย)
  if (old.score_type !== i.score_type) {
    const [[{ n }]] = await pool.query(
      `SELECT (SELECT COUNT(*) FROM self_evaluations WHERE indicator_id = ? AND score IS NOT NULL)
            + (SELECT COUNT(*) FROM committee_scores WHERE indicator_id = ?) AS n`,
      [req.params.id, req.params.id]
    );
    if (n) return fail(res, 400, "ตัวชี้วัดนี้มีการให้คะแนนแล้ว เปลี่ยนรูปแบบการประเมินไม่ได้");
  }

  const [result] = await pool.query(
    "UPDATE indicators SET topic_id = ?, name = ?, description = ?, weight = ?, score_type = ?, evidence_type = ?, evidence_note = ? WHERE id = ?",
    [i.topic_id, i.name, i.description, i.weight, i.score_type, i.evidence_type, i.evidence_note, req.params.id]
  );
  if (!result.affectedRows) return fail(res, 404, "ไม่พบตัวชี้วัด");
  ok(res, null, "แก้ไขตัวชี้วัดสำเร็จ");
});

import StatusBadge from "@/components/StatusBadge";

export default function Assignments() {
  const user = useAuth(["hr"]);
  const [evaluatees, setEvaluatees] = useState([]);
  const [committees, setCommittees] = useState([]);
  const [assignments, setAssignments] = useState([]);
  const [indicatorCount, setIndicatorCount] = useState(0);
  const [error, setError] = useState("");
  const [form, setForm] = useState({}); // ค่าที่เลือกในแถวเพิ่มกรรมการ แยกตาม evaluatee_id

  // โหลด 4 อย่างพร้อมกัน
  const load = () =>
    Promise.all([
      api.get("/users", { params: { role: "evaluatee" } }),
      api.get("/users", { params: { role: "committee" } }),
      api.get("/assignments"),
      api.get("/indicators"),
    ]).then(([e, c, a, i]) => {
      setEvaluatees(e.data.data);
      setCommittees(c.data.data);
      setAssignments(a.data.data);
      setIndicatorCount(i.data.data.length);
    }).catch((err) => setError(errorMessage(err)));

  useEffect(() => {
    if (user) load();
  }, [user]);

  // เรียก API แล้วโหลดใหม่ ถ้า error แสดง alert
  const run = async (request) => {
    try {
      await request;
      load();
    } catch (err) {
      alert(errorMessage(err));
    }
  };

  const add = (evaluateeId) => {
    const f = form[evaluateeId] || {};
    if (!f.committee_id) return alert("กรุณาเลือกกรรมการ");
    run(
      api.post("/assignments", {
        evaluatee_id: evaluateeId,
        committee_id: f.committee_id,
        committee_role: f.committee_role || "member",
      })
    );
    setForm({ ...form, [evaluateeId]: {} });
  };

  const setField = (evaluateeId, key, value) =>
    setForm({ ...form, [evaluateeId]: { ...form[evaluateeId], [key]: value } });

  if (!user) return null;

  return (
    <Layout user={user}>
      {error && <div className="alert alert-error mb-4">{error}</div>}
      <h1 className="text-2xl font-bold mb-1">มอบหมายกรรมการ</h1>
      <p className="text-sm opacity-70 mb-4">กำหนดกรรมการและบทบาท (ประธาน 1 คนต่อผู้รับการประเมิน) และติดตามสถานะการประเมิน</p>

      <div className="grid lg:grid-cols-2 gap-4">
        {evaluatees.map((e) => {
          const mine = assignments.filter((a) => a.evaluatee_id === e.id);
          const available = committees.filter((c) => !mine.some((a) => a.committee_id === c.id));
          const f = form[e.id] || {};
          return (
            <div key={e.id} className="card bg-base-100 shadow">
              <div className="card-body p-4">
                <h2 className="card-title text-base">
                  {e.full_name}
                  <span className="text-sm font-normal opacity-60">{e.department}</span>
                </h2>

                <ul className="divide-y divide-base-200">
                  {mine.map((a) => (
                    <li key={a.id} className="py-2 flex flex-wrap items-center gap-2">
                      <span className="flex-1 min-w-32">{a.committee_name}</span>
                      <select
                        className="select select-sm w-36"
                        value={a.committee_role}
                        onChange={(ev) => run(api.put(`/assignments/${a.id}`, { committee_role: ev.target.value }))}
                      >
                        {Object.entries(COMMITTEE_ROLES).map(([value, label]) => (
                          <option key={value} value={value}>{label}</option>
                        ))}
                      </select>
                      <StatusBadge status={a.status} />
                      <span className="text-xs opacity-60 w-16">
                        {a.scored_count}/{indicatorCount} ข้อ
                      </span>
                      <button
                        className="btn btn-xs btn-ghost text-error"
                        title="ยกเลิกการมอบหมาย"
                        onClick={() => confirm(`ยกเลิก ${a.committee_name} ? คะแนนที่ให้ไว้จะถูกลบ`) && run(api.delete(`/assignments/${a.id}`))}
                      >
                        <Trash2 size={14} />
                      </button>
                    </li>
                  ))}
                  {mine.length === 0 && <li className="py-2 text-sm opacity-60">ยังไม่มีกรรมการ</li>}
                </ul>

                {available.length > 0 && (
                  <div className="flex flex-wrap gap-2 mt-1">
                    <select
                      className="select select-sm flex-1 min-w-40"
                      value={f.committee_id || ""}
                      onChange={(ev) => setField(e.id, "committee_id", ev.target.value)}
                    >
                      <option value="">เลือกกรรมการ</option>
                      {available.map((c) => (
                        <option key={c.id} value={c.id}>{c.full_name}</option>
                      ))}
                    </select>
                    <select
                      className="select select-sm w-36"
                      value={f.committee_role || "member"}
                      onChange={(ev) => setField(e.id, "committee_role", ev.target.value)}
                    >
                      {Object.entries(COMMITTEE_ROLES).map(([value, label]) => (
                        <option key={value} value={value}>{label}</option>
                      ))}
                    </select>
                    <button className="btn btn-sm btn-primary" onClick={() => add(e.id)}>
                      <Plus size={16} /> เพิ่ม
                    </button>
                  </div>
                )}
              </div>
            </div>
          );
        })}
      </div>
    </Layout>
  );
}
