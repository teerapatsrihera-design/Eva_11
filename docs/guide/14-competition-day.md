### ขั้นที่ 14.3.1 Backend: route CRUD (บท 03-05)

> 🎯 **ขั้นนี้จะได้:** แม่แบบ route 5 ตัวที่ใช้ซ้ำได้ทุกตาราง

```js
import { Router } from "express";
import pool from "../db.js";
import { ok, fail } from "../utils/response.js";
import { verifyToken, requireRole } from "../middleware/auth.js";
import { validate } from "../middleware/validate.js";
import { xSchema } from "../schemas/x.js";

const router = Router();
router.use(verifyToken, requireRole("hr"));

router.get("/", async (req, res) => {                                   // อ่านทั้งหมด
  const [rows] = await pool.query("SELECT * FROM x ORDER BY id");
  ok(res, rows);
});
router.get("/:id", async (req, res) => {                                // อ่าน 1
  const [rows] = await pool.query("SELECT * FROM x WHERE id = ?", [req.params.id]);
  if (!rows[0]) return fail(res, 404, "ไม่พบข้อมูล");
  ok(res, rows[0]);
});
router.post("/", validate(xSchema), async (req, res) => {               // เพิ่ม
  const [result] = await pool.query("INSERT INTO x (a, b) VALUES (?, ?)", [req.body.a, req.body.b]);
  ok(res, { id: result.insertId }, "เพิ่มสำเร็จ", 201);
});
router.put("/:id", validate(xSchema), async (req, res) => {             // แก้
  const [result] = await pool.query("UPDATE x SET a = ?, b = ? WHERE id = ?", [req.body.a, req.body.b, req.params.id]);
  if (!result.affectedRows) return fail(res, 404, "ไม่พบข้อมูล");
  ok(res, null, "แก้ไขสำเร็จ");
});
router.delete("/:id", async (req, res) => {                             // ลบ
  const [result] = await pool.query("DELETE FROM x WHERE id = ?", [req.params.id]);
  if (!result.affectedRows) return fail(res, 404, "ไม่พบข้อมูล");
  ok(res, null, "ลบสำเร็จ");
});

export default router;
// อย่าลืม: app.use("/api/x", xRoutes) ใน app.js
```

| ส่วน | จำ |
|---|---|
| ด่าน | `router.use(verifyToken, requireRole(...))` บนสุด (บท 04 ขั้น 4.7) |
| อ่าน 1 / แก้ / ลบ ไม่พบ | 404 |
| เพิ่มสำเร็จ | **201** + `{ id: result.insertId }` |
| ตรวจ body | `validate(schema)` ก่อน async |

### ขั้นที่ 14.3.2 Frontend: หน้า CRUD (บท 08)

> 🎯 **ขั้นนี้จะได้:** แม่แบบหน้ารายการ + ปุ่มเพิ่ม/ลบ + Modal

```jsx
import { useEffect, useState } from "react";
import api, { errorMessage } from "@/lib/api";
import { useAuth } from "@/lib/auth";
import Layout from "@/components/Layout";
import XModal from "@/components/XModal";

export default function XPage() {
  const user = useAuth(["hr"]);
  const [items, setItems] = useState([]);
  const [error, setError] = useState("");
  const [editing, setEditing] = useState(null);          // null ปิด, {} เพิ่ม, item แก้

  const load = () => api.get("/x").then((res) => setItems(res.data.data)).catch((err) => setError(errorMessage(err)));
  useEffect(() => { if (user) load(); }, [user]);

  const remove = async (item) => {
    if (!confirm(`ลบ ${item.name} ?`)) return;
    try { await api.delete(`/x/${item.id}`); load(); } catch (err) { alert(errorMessage(err)); }
  };

  if (!user) return null;
  return (
    <Layout user={user}>
      {error && <div className="alert alert-error mb-4">{error}</div>}
      <button className="btn btn-primary" onClick={() => setEditing({})}>เพิ่ม</button>
      <div className="overflow-x-auto"><table className="table">...{items.map((i) => <tr key={i.id}>...</tr>)}</table></div>
      {editing && <XModal item={editing} onClose={() => setEditing(null)} onSaved={() => { setEditing(null); load(); }} />}
    </Layout>
  );
}
```

| ส่วน | จำ |
|---|---|
| `editing` | `null` = ปิด Modal, `{}` = เพิ่ม, ข้อมูล = แก้ (บท 08 ขั้น 8.1) |
| `load` | ทุก `.then` ต้องมี `.catch` |
| `if (!user) return null` | ก่อน return JSX เสมอ |
| ตาราง | ครอบ `overflow-x-auto` (มือถือ) |

### ขั้นที่ 14.3.3 Frontend: Modal ฟอร์ม (บท 08)

> 🎯 **ขั้นนี้จะได้:** แม่แบบฟอร์มเพิ่ม/แก้ด้วย React Hook Form + Zod

```jsx
export default function XModal({ item, onClose, onSaved }) {
  const [error, setError] = useState("");
  const { register, handleSubmit, formState: { errors, isSubmitting } } = useForm({
    resolver: validateWith(xSchema),
    defaultValues: formValues({ name: "", ...item }),
  });
  const onSubmit = async (data) => {
    try {
      if (item.id) await api.put(`/x/${item.id}`, data);
      else await api.post("/x", data);
      onSaved();
    } catch (err) { setError(errorMessage(err)); }
  };
  return (
    <Modal title={item.id ? "แก้ไข" : "เพิ่ม"} onClose={onClose}>
      <form onSubmit={handleSubmit(onSubmit)}>
        {error && <div className="alert alert-error">{error}</div>}
        <Field label="ชื่อ" error={errors.name}><input className="input w-full" {...register("name")} /></Field>
        <div className="modal-action">
          <button type="button" className="btn" onClick={onClose}>ยกเลิก</button>
          <button className="btn btn-primary" disabled={isSubmitting}>บันทึก</button>
        </div>
      </form>
    </Modal>
  );
}
```

| ส่วน | จำ |
|---|---|
| `resolver: validateWith(schema)` | ห้ามใช้ @hookform/resolvers (บท 07 หัวข้อ 7.7) |
| `formValues(...)` | แปลง null เป็น `""` กัน warning controlled input (บท 07 ขั้น 7.11.2) |
| `item.id` มีไหม | แยกเพิ่ม (POST) / แก้ (PUT) ด้วยฟอร์มเดียว |
| ปุ่มยกเลิก | `type="button"` ไม่งั้นส่งฟอร์ม |

### ขั้นที่ 14.3.4 จำ 10 ข้อนี้ให้ได้

> 🎯 **ขั้นนี้จะได้:** กันความผิดพลาดที่เสียเวลาที่สุด

1. import ไฟล์ตัวเองใน backend ต้องมี `.js`
2. SQL ใช้ `?` เสมอ ห้ามต่อ string
3. ทุก route (ยกเว้น login) มี `verifyToken` + `requireRole`
4. response ใช้ `ok()` / `fail()` เท่านั้น
5. POST สำเร็จตอบ **201**, ไม่พบตอบ **404**
6. ห้าม `SELECT *` จากตาราง users (มี password_hash)
7. localStorage ใช้ใน useEffect เท่านั้น
8. ทุกหน้าที่ต้อง login: `const user = useAuth([...]); if (!user) return null;`
9. ทุก `api.get().then()` ต้องมี `.catch()`
10. `globals.css` ต้องมี `exclude: join` (บท 07)

### ขั้นที่ 14.4.1 Flow Diagram 5 แผ่น (ส่วนที่ 1)

> 🎯 **ขั้นนี้จะได้:** ผังงานที่ตรงกับระบบจริง

**1.1 Login Flow**

```text
([เริ่ม])
   ↓
[/กรอก username, password/]  ◀──────────────────┐
   ↓                                             │
<กรอกครบ?> ── ไม่ ──────────────────────────────▶┤
   ↓ ครบ                                         │
<username/password ถูก?> ── ไม่ ──▶ [แสดง: ชื่อผู้ใช้หรือรหัสผ่านไม่ถูกต้อง]
   ↓ ถูก
[ออก JWT token เก็บใน localStorage]
   ↓
<ต้องเปลี่ยนรหัส?> ── ใช่ ──▶ [หน้าเปลี่ยนรหัสผ่าน]
   ↓ ไม่                          ↓
<role?> ◀─────────────────────────┘
   ├─ hr ─────────▶ [Dashboard งานบุคลากร]
   ├─ evaluatee ──▶ [หน้าหลักผู้รับการประเมิน]
   └─ committee ──▶ [หน้าหลักกรรมการ]
   ↓
([จบ])
```

| สัญลักษณ์ | ความหมาย |
|---|---|
| `([ ])` วงรี | เริ่ม / จบ |
| `[/ /]` สี่เหลี่ยมด้านขนาน | รับข้อมูล / แสดงผล |
| `[ ]` สี่เหลี่ยม | การทำงาน |
| `< >` สี่เหลี่ยมขนมเปียกปูน | ตัดสินใจ (มีทางออก 2 ทาง) |

**1.2 งานบุคลากร:** Login → สร้างหัวข้อ + กำหนดวันเปิด-ปิด → เพิ่มตัวชี้วัด (ชื่อ, น้ำหนัก, รูปแบบ, หลักฐาน) → เพิ่มผู้ใช้ (ได้รหัสชั่วคราว) → มอบหมายกรรมการ + บทบาท → ติดตามสถานะ → ดูรายงาน

**1.3 ผู้รับการประเมิน:** Login → (ครั้งแรก) เปลี่ยนรหัส → เลือกตัวชี้วัด → ให้คะแนนตนเอง → แนบหลักฐาน (ไฟล์/URL) → ครบทุกตัว? → ดูผล + ความเห็นกรรมการ → Export PDF

**1.4 กรรมการ:** Login → เลือกผู้รับการประเมิน → ดูหลักฐาน + คะแนนตนเอง → ให้คะแนน → ความเห็นสรุป → บันทึกร่าง → ครบ? → ลงลายเซ็น → ยืนยัน (ล็อก) (บท 10 ขั้น 10.1.1)

### ขั้นที่ 14.6.2 ซ้อมตอบคำถาม (เกณฑ์ 9.1, 9.2)

> 🎯 **ขั้นนี้จะได้:** ตอบได้ทันทีพร้อมชี้โค้ดจริง

**Backend**

| คำถาม | แนวตอบ | ชี้โค้ด |
|---|---|---|
| JWT คืออะไร ปลอดภัยอย่างไร | token ที่มีลายเซ็นจาก secret ฝั่ง server แก้ข้อมูลแล้วลายเซ็นไม่ตรง ตรวจด้วย `jwt.verify` หมดอายุใน 8 ชั่วโมง | บท 04 |
| 401 กับ 403 ต่างกันอย่างไร | 401 = ยังไม่ได้ยืนยันตัวตน (ไม่มี/ผิด token), 403 = รู้ว่าเป็นใครแล้วแต่ไม่มีสิทธิ์ | บท 04 ขั้น 4.5 |
| เก็บรหัสผ่านอย่างไร | hash ด้วย crypto.scrypt + salt สุ่ม เก็บแบบ salt:hash ตอน login hash รหัสที่กรอกแล้วเทียบด้วย timingSafeEqual | บท 04 ขั้น 4.2 |
| ป้องกัน SQL Injection อย่างไร | ใช้ placeholder `?` ทุกคำสั่ง mysql2 escape ให้ ไม่ต่อ string | บท 03 |
| ทำไมต้อง validate ทั้ง 2 ฝั่ง | ฝั่งหน้าเพื่อความสะดวกของผู้ใช้ ฝั่ง backend เพื่อความปลอดภัย เพราะยิง API ตรงได้โดยไม่ผ่านหน้าเว็บ | บท 04, 07 |
| คำนวณคะแนนอย่างไร | (คะแนน ÷ คะแนนเต็ม) × น้ำหนัก สเกลเต็ม 4 มี/ไม่มีเต็ม 1 ใช้ค่าเฉลี่ยกรรมการที่ยืนยันแล้ว | บท 06 ขั้น 6.3 |
| RESTful คืออะไร | ใช้ URL เป็นคำนามแทนทรัพยากร ใช้ HTTP method บอกการกระทำ ตอบ status code ตรงความหมาย | บท 13 ขั้น 13.2 |
| Middleware คืออะไร | ฟังก์ชันที่คำขอผ่านก่อนถึง route เช่น verifyToken, requireRole, validate | บท 04 |
| อัปโหลดไฟล์ปลอดภัยอย่างไร | จำกัดชนิด (นามสกุลต้องตรง mimetype) ขนาด 10MB ตั้งชื่อสุ่ม ดาวน์โหลดผ่าน API ที่ตรวจสิทธิ์ | บท 06 ขั้น 6.4 |
| Transaction ใช้ตรงไหน ทำไม | บันทึกคะแนนกรรมการ + เปลี่ยนสถานะพร้อมกัน ถ้าพังกลางทางย้อนกลับทั้งหมด | บท 06 ขั้น 6.7.7 |
| Backup/Restore ทำอย่างไร | mysqldump ได้ไฟล์ .sql, กู้คืนส่งไฟล์เข้า mysql, สำรองข้อมูลปัจจุบันก่อนกู้คืนเสมอ | บท 11 ขั้น 11.4 |

**Frontend**

| คำถาม | แนวตอบ | ชี้โค้ด |
|---|---|---|
| useState กับ useEffect ใช้ทำอะไร | useState เก็บข้อมูลที่เปลี่ยนแล้วหน้าจออัปเดต, useEffect ทำงานหลังแสดงผล เช่น โหลดข้อมูล | บท 07 |
| ทำไมอ่าน localStorage ใน useEffect | Next.js render บน server ก่อน ซึ่งไม่มี localStorage, useEffect ทำงานบนเบราว์เซอร์เท่านั้น | บท 07 |
| ป้องกันหน้าที่ต้อง login อย่างไร | useAuth ตรวจ user/role ใน useEffect ไม่ผ่านพาไปหน้าอื่น แต่ความปลอดภัยจริงอยู่ที่ backend | บท 07 |
| ทำ PDF อย่างไรโดยไม่ใช้ jspdf | `window.print()` + class `print:hidden` ซ่อนส่วนที่ไม่ต้องพิมพ์ แล้วเลือกบันทึกเป็น PDF | บท 09 |
| ลายเซ็นทำอย่างไร | canvas + pointer events วาดเส้นตามเมาส์/นิ้ว แปลงเป็นรูปด้วย `toDataURL` เก็บเป็นข้อความในฐานข้อมูล | บท 10 ขั้น 10.3 |
| Responsive ทำอย่างไร | Tailwind breakpoint sm/md/lg, เมนู dropdown บนมือถือ, ตารางครอบ overflow-x-auto, flex-wrap | บท 12 ขั้น 12.11.1 |
| Dashboard ไม่ใช้ chart library ทำอย่างไร | ใช้ stat, radial-progress, progress ของ daisyUI | บท 11 ขั้น 11.3 |
